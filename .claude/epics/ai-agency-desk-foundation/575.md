class DeskResponse(BaseModel):
    """Response from a single desk agent.

    Frozen (immutable after construction) -- represents a completed desk output.
    ``confidence`` is validated to be finite and within [0.0, 1.0].
    """

    model_config = ConfigDict(frozen=True)

    desk: DeskType
    response: str
    tools_used: list[str]
    confidence: float  # 0.0 to 1.0

    @field_validator("confidence")
    @classmethod
    def validate_confidence(cls, v: float) -> float:
        """Ensure confidence is finite and within [0.0, 1.0]."""
        return validate_unit_interval(v, "confidence")

### AgencyConfig (config.py)

```python
class AgencyConfig(BaseModel):
    """AI agency desk system configuration.

    Controls timeouts and tool budgets for desk agents. Tool budget limits
    how many tool calls a desk agent can make per query to control cost and
    latency. Risk and research desks get higher budgets by default.
    """

    agent_timeout: float = 60.0
    default_tool_budget: int = 3
    risk_tool_budget: int = 5
    research_tool_budget: int = 5

    @field_validator("agent_timeout")
    @classmethod
    def validate_agent_timeout(cls, v: float) -> float:
        """Ensure agent_timeout is finite and positive."""
        if not math.isfinite(v):
            raise ValueError(f"agent_timeout must be finite, got {v}")
        if v <= 0.0:
            raise ValueError(f"agent_timeout must be > 0, got {v}")
        return v

    @field_validator("default_tool_budget", "risk_tool_budget", "research_tool_budget")
    @classmethod
    def validate_tool_budget(cls, v: int) -> int:
        """Ensure tool budgets are within [1, 20]."""
        if not 1 <= v <= 20:
            raise ValueError(f"tool_budget must be in [1, 20], got {v}")
        return v
```

### Edge Cases

- `DeskResponse` with confidence exactly `0.0` -- valid lower boundary
- `DeskResponse` with confidence exactly `1.0` -- valid upper boundary
- `DeskResponse` with confidence `NaN`, `Inf`, `-Inf` -- all rejected by `isfinite()` check
- `DeskResponse` with confidence `-0.1` and `1.1` -- rejected by range check
- `QueryIntent` with empty `tickers` list -- valid (query may not reference specific tickers)
- `AgencyConfig` with all default values -- valid production config
- `AgencyConfig` with `agent_timeout=0.0` -- rejected (must be positive)
- `AgencyConfig` with `default_tool_budget=0` -- rejected (must be >= 1)
- `AppSettings` with `ARENA_AGENCY__AGENT_TIMEOUT=90` env var -- overrides default

### Mocking Strategy

No mocking needed. These are pure data models -- construct and validate directly.
`monkeypatch.setenv()` for the AppSettings env var override test.
