class AgencyQuery(BaseModel):
    """A user query submitted to the agency routing system.

    Frozen (immutable after construction) -- represents a completed query submission.
    ``query_id`` is a UUID4 string assigned at creation time.
    """

    model_config = ConfigDict(frozen=True)

    query_id: str
    query_text: str
    created_at: datetime
    desk_override: DeskType | None = None

    @field_validator("created_at")
    @classmethod
    def _validate_utc(cls, v: datetime) -> datetime:
        if v.tzinfo is None or v.utcoffset() != timedelta(0):
            raise ValueError("created_at must be UTC")
        return v
```

### AgencyResponse Model (models/analysis.py)

class AgencyResponse(BaseModel):
    """Synthesized response from the agency routing system.

    Frozen (immutable after construction). Combines desk responses into a single
    coherent output with merged citations and weighted confidence.
    """

    model_config = ConfigDict(frozen=True)

    query_id: str
    query_text: str
    intent: QueryIntent
    desk_responses: list[DeskResponse]
    synthesis: str
    citations: list[Citation]
    confidence: float

    created_at: datetime

    @field_validator("confidence")
    @classmethod
    def _validate_confidence(cls, v: float) -> float:
        return validate_unit_interval(v, "confidence")

    @field_validator("created_at")
    @classmethod
    def _validate_utc(cls, v: datetime) -> datetime:
        if v.tzinfo is None or v.utcoffset() != timedelta(0):
            raise ValueError("created_at must be UTC")
        return v
```

### classify_intent (agents/_routing.py)

```python
def classify_intent(query: str) -> QueryIntent:
    """Classify a natural-language query into desk routing intent.

    Rule-based V1 -- keyword matching + regex ticker extraction.
    No LLM call. Returns QueryIntent with desk(s), query type, and tickers.

    Keyword -> desk mapping:
      - "volatility", "iv", "vega", "implied vol", "vol surface" -> VOLATILITY
      - "risk", "hedge", "exposure", "drawdown", "var" -> RISK
      - "trend", "momentum", "moving average", "sma", "ema", "macd" -> TREND
      - "flow", "volume", "open interest", "unusual activity" -> FLOW
      - "fundamental", "earnings", "p/e", "revenue", "valuation" -> FUNDAMENTAL
      - "contrarian", "sentiment", "overcrowded", "reversal" -> CONTRARIAN

    Ticker extraction:
      - $AAPL syntax -> "AAPL"
      - Standalone uppercase 1-5 letter words matching TICKER_RE
      - Excludes common English words: IV, PE, GDP, etc.

    Query type inference:
      - "compare", "vs", "versus" -> COMPARISON
      - "strategy", "recommend", "trade" -> STRATEGY
      - "risk", "hedge", "exposure" -> RISK_CHECK
      - "analyze", "analysis", "what" -> ANALYSIS
      - Default: GENERAL

    Default desk: VOLATILITY (most general) when no keywords match.
    """
```

### run_agency_query (agents/_routing.py)

```python
async def run_agency_query(
    query: AgencyQuery,
    *,
    market_data: MarketDataService,
    options_data: OptionsDataService,
    fred: FredService,
    repo: Repository,
    model: Model | None,
    config: AgencyConfig,
) -> AgencyResponse:
    """Route a user query to desk agent(s) and synthesize the response.

    Orchestration flow:
    1. classify_intent(query.query_text) -> QueryIntent
    2. Override desk if query.desk_override is set
    3. Dispatch to desk(s) via asyncio.gather(*coroutines, return_exceptions=True)
    4. For implemented desks (vol, risk): call run_vol_desk_query / run_risk_desk_query
    5. For unimplemented desks: return error DeskResponse(confidence=0.0)
    6. Synthesize AgencyResponse with merged citations, weighted confidence
    7. Never raises -- catches all exceptions, returns error AgencyResponse
    """
```

### Edge Cases

- `classify_intent("")` -- empty query defaults to VOLATILITY desk, GENERAL type, no tickers
- `classify_intent("IV")` -- "IV" is both a keyword and a non-ticker word; routes to VOLATILITY, no ticker extracted
- `classify_intent("$AAPL $AAPL")` -- duplicate tickers deduplicated
- `classify_intent("aapl volatility")` -- lowercase "aapl" should still be extracted as ticker (case-insensitive)
- `classify_intent("What about BRK-B?")` -- hyphenated tickers matching TICKER_RE
- `AgencyQuery` with naive datetime -- rejected by UTC validator
- `AgencyQuery` with EST timezone -- rejected by UTC validator
- `AgencyResponse` with confidence=NaN/Inf -- rejected by isfinite check
- `AgencyResponse` with empty desk_responses -- valid (all desks failed)
- `run_agency_query` with classify_intent raising -- caught, returns error response
- `run_agency_query` with desk runner raising -- caught per-desk, other desks still return
- `run_agency_query` with model=None -- desk runners handle None model gracefully

### Mocking Strategy

- `MagicMock()` for service instances (market_data, options_data, fred, repo)
- `AsyncMock` for patching `run_vol_desk_query` / `run_risk_desk_query`
- `patch()` for isolating classify_intent in run_agency_query error tests
- No real service connections, no database, no network calls
- No `pydantic_ai.models.ALLOW_MODEL_REQUESTS = False` needed (desk runners are patched)
