class TestRecommendationExport:
    def test_basic_export(self, sample_recommendation_result: RecommendationResult) -> None:
        """Verify markdown export contains key sections."""
        md = export_recommendation_markdown(sample_recommendation_result)
        assert "## Position Recommendation" in md
        assert "## Domain Assessments" in md

    def test_all_assessments_rendered(self, sample_recommendation_result: RecommendationResult) -> None:
        """Verify all 6 domain assessments appear in output."""
        md = export_recommendation_markdown(sample_recommendation_result)
        for desk in ["Trend", "Volatility", "Flow", "Fundamental", "Risk", "Contrarian"]:
            assert desk in md

    def test_decimal_precision_preserved(self, sample_recommendation_result: RecommendationResult) -> None:
        """Verify Decimal fields rendered with 2 decimal places."""
        md = export_recommendation_markdown(sample_recommendation_result)
        # Strike price should appear formatted
        assert "185.00" in md or "185.50" in md

    def test_fallback_badge(self, sample_recommendation_result_fallback: RecommendationResult) -> None:
        """Verify fallback indicator appears when is_fallback=True."""
        md = export_recommendation_markdown(sample_recommendation_result_fallback)
        assert "Fallback" in md or "fallback" in md

    def test_direction_displayed(self, sample_recommendation_result: RecommendationResult) -> None:
        """Verify direction (BULLISH/BEARISH/NEUTRAL) is shown."""
        md = export_recommendation_markdown(sample_recommendation_result)
        assert any(d in md for d in ["BULLISH", "BEARISH", "NEUTRAL", "Bullish", "Bearish", "Neutral"])

    def test_empty_assessments_handled(self) -> None:
        """Verify graceful handling when assessments list is empty."""
        # Build result with empty assessments
        result = build_minimal_recommendation_result(assessments=[])
        md = export_recommendation_markdown(result)
        assert "No assessments" in md or "## Position Recommendation" in md
