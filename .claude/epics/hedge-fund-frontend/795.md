## Technical Details

### types/recommendation.ts — New interfaces

```typescript
type PipelineStage = 'queued' | 'scored' | 'analyzing' | 'ready' | 'failed'

interface PipelineTicker {
  ticker: string
  composite_score: number
  direction: 'BULLISH' | 'BEARISH' | 'NEUTRAL'
  direction_confidence: number | null
  dimensional_scores: DimensionalScores | null  // reuse from scan.ts
  sector: string | null
  company_name: string | null
  stage: PipelineStage
  recommendation_id: number | null
}

interface DeskAssessment {
  desk: string
  direction: string
  confidence: number
  summary: string
  key_findings: string[]
}

interface PositionRecommendation {
  ticker: string
  option_type: string | null
  strike: string | null          // Decimal as string — NEVER parse to number
  expiration: string | null
  recommended_contract: string
  entry_price: string
  stop_loss: string | null
  take_profit: string | null
  position_size_pct: number
  risk_reward_ratio: number
  direction: string
  confidence: number
  strategy: string | null
  strategy_rationale: string
  rationale: string
}

interface RecommendationDetail {
  id: number
  ticker: string
  assessments: DeskAssessment[]
  recommendation: PositionRecommendation
  is_fallback: boolean
  recommendation_protocol: string
  duration_ms: number
  total_tokens: number
  citation_density: number
  model_used: string
  created_at: string
  scan_run_id: number | null
}

interface PredictionAccuracy {
  source: string
  total: number
  correct: number
  accuracy: number
  sample_sufficient: boolean
}

interface ConditionBucketAccuracy {
  source: string
  condition: string
  total: number
  correct: number
  accuracy: number
}

interface ContractGuidance {
  optimal_delta_low: number
  optimal_delta_high: number
  optimal_dte_low: number
  optimal_dte_high: number
  delta_win_rate: number
  dte_win_rate: number
  sample_count: number
}

interface AttributionReport {
  window_days: number
  total_recommendations: number
  total_outcomes: number
  source_accuracy: PredictionAccuracy[]
  condition_accuracy: ConditionBucketAccuracy[]
  contract_guidance: ContractGuidance | null
}
```

### types/scan.ts — UPPERCASE direction fix

Update `TickerScore.direction` from `'bullish' | 'bearish' | 'neutral'` to `'BULLISH' | 'BEARISH' | 'NEUTRAL'`. Same for `RecommendedContract.direction` and `MarketRegime` values if applicable. Update `ScanPreset` type union to match backend casing.

### types/index.ts — Barrel export update

Remove `export * from './debate'`, add `export * from './recommendation'`.

### stores/pipeline.ts — State machine

Setup-syntax Pinia store with:
- **State**: `tickers: Map<string, PipelineTicker>`, `selectedTicker: string | null`, `currentRecommendation: RecommendationDetail | null`, `scanId: number | null`, `batchId: number | null`, `phase: 'idle' | 'scanning' | 'scanned'`, `loading: boolean`
- **Getters**: `sortedTickers` (by composite_score desc), `selectedPipelineTicker`, `hasReadyTickers`, `scanProgress`
- **Actions**: `startScan(options)`, `analyzeTicker(ticker)`, `analyzeBatch(tickers)`, `selectTicker(ticker)`, `loadRecommendation(debateId)`, `reset()`
- **WS callbacks**: `onScanProgress(event)`, `onScanComplete(event)`, `onDebateComplete(ticker, debateId)`, `onBatchProgress(event)`, `onBatchComplete(event)`
