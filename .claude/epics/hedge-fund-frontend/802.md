### WebSocket wiring
```typescript
// Scan WS — connect when scan starts
function connectScanWs(scanId: number): void {
  const { send, close } = useWebSocket<ScanEvent>({
    url: `/ws/scan/${scanId}`,
    onMessage(event) {
      switch (event.type) {
        case 'progress': pipelineStore.onScanProgress(event); break
        case 'complete': pipelineStore.onScanComplete(event); closeScanWs(); break
        case 'error': toast.add({ severity: 'error', summary: 'Scan Error', detail: event.message }); break
      }
    }
  })
  scanWsSend.value = send
  scanWsClose.value = close
}

// Single debate WS — connect when single ticker analyzed
function connectDebateWs(debateId: number, ticker: string): void { ... }

// Batch WS — connect when batch analyzed
function connectBatchWs(batchId: number): void { ... }
```

### Action handlers
- `onAnalyzeTicker(ticker)` → `POST /api/debate { ticker, scan_id }` → connect debate WS → update stage to `analyzing`
- `onAnalyzeSelected()` → `POST /api/debate/batch { scan_id, tickers }` → connect batch WS → update stages
- `onAnalyzeTopN(limit)` → `POST /api/debate/batch { scan_id, limit }` → connect batch WS
- `onSelectTicker(ticker)` → if stage is `ready`, `GET /api/debate/{id}` → load recommendation detail
