# Task: Update DebateConfig — remove dead fields, add new fields

Clean up `DebateConfig` in `models/config.py` to remove 4 dead fields from the old debate system and add 5 new fields for the recommendation system. Rename `min_debate_score` to `min_recommendation_score`. This is foundational — other cutover tasks depend on the new config shape.

- `synthesis_timeout: float = 90.0` — seconds for synthesis agent
- `recommendation_protocol: str = "unified_v1"` — tags new vs old records in analytics
- `min_recommendation_score: float = 30.0` — replaces `min_debate_score`

- `disabled_desks: list[str] = []` — desks to skip (config-driven)
- All new fields need `math.isfinite()` validators where numeric (NaN defense rule)

```python
class TestDebateConfigCutover:
    def test_dead_fields_removed(self) -> None:
        """Verify removed fields are not accepted."""
        with pytest.raises(ValidationError):
            DebateConfig(enable_volatility_agent=True)

    def test_new_fields_defaults(self) -> None:
        """Verify new fields have correct defaults."""
        config = DebateConfig()
        assert config.synthesis_timeout == 90.0
        assert config.recommendation_protocol == "unified_v1"
        assert config.min_recommendation_score == 30.0
        assert config.desk_parallelism == 6
        assert config.disabled_desks == []

    def test_synthesis_timeout_nan_rejected(self) -> None:
        """Verify NaN defense on synthesis_timeout."""
        with pytest.raises(ValidationError):
            DebateConfig(synthesis_timeout=float("nan"))

    def test_min_recommendation_score_nan_rejected(self) -> None:
        """Verify NaN defense on min_recommendation_score."""
        with pytest.raises(ValidationError):
            DebateConfig(min_recommendation_score=float("inf"))

    def test_desk_parallelism_range(self) -> None:
        """Verify desk_parallelism bounded [1, 12]."""
        with pytest.raises(ValidationError):
            DebateConfig(desk_parallelism=0)
        with pytest.raises(ValidationError):
            DebateConfig(desk_parallelism=13)

    def test_env_var_prefix_preserved(self) -> None:
        """Verify ARENA_DEBATE__ prefix still works via AppSettings."""
        # Construct AppSettings with debate sub-config
        settings = AppSettings(debate=DebateConfig(desk_parallelism=4))
        assert settings.debate.desk_parallelism == 4

    def test_disabled_desks_accepts_list(self) -> None:
        """Verify disabled_desks accepts string list."""
        config = DebateConfig(disabled_desks=["risk", "contrarian"])
        assert len(config.disabled_desks) == 2
```

### Edge Cases
- NaN/Inf in numeric fields: rejected by `isfinite()` validator
- `desk_parallelism=0`: rejected (would cause deadlock)
- Empty `disabled_desks`: valid default
- `min_recommendation_score=0.0`: valid (disables score gate)
